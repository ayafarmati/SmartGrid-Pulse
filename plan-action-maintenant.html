# 1. Initialiser Dim_Date et Dim_Time (2015-2018)
bq --project_id=smartgrid-pulse --location=europe-west1 query \
  --use_legacy_sql=false --maximum_bytes_billed=104857600 \
  --parameter='start_date:DATE:2015-01-01' \
  --parameter='end_date:DATE:2018-12-31' < sql/seed_calendar.sql

# 2. Construire et pousser l'image Docker du replay 2018 (Cloud Shell)
repo_uri=$("$HOME/.local/bin/terraform" output -raw simulator_repository)
replay_image="$repo_uri/business-replay:v0.1.0"
docker build --platform linux/amd64 -t "$replay_image" -f replay/Dockerfile .
docker push "$replay_image"
replay_digest=$(docker image inspect --format '{{index .RepoDigests 0}}' "$replay_image")

# 3. Activer business_pipeline.tf (plan obligatoire avant apply !)
cat <<EOF > business-pipeline.auto.tfvars
enable_business_pipeline  = true
business_replay_image     = "$replay_digest"
business_replay_input_uri = "gs://smartgrid-pulse-raw-bronze/replay-input/2018/events.ndjson"
EOF
"$HOME/.local/bin/terraform" validate && "$HOME/.local/bin/terraform" plan -out=smartgrid-business.tfplan
